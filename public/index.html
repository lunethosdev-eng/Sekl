Guía de Integración para Clientes (Web, Android, Roku)

Para evitar dependencias de URLs directas y permitir la reproducción offline, los clientes deben solicitar el stream binario y persistirlo localmente.

1. JavaScript / Web / WebView (Guardado Automático en IndexedDB)


// Función para descargar y guardar el archivo MP3 directamente en IndexedDB
async function downloadAndCacheAudio(trackId, audioUrl) {
  const db = await openDB();
  const response = await fetch(audioUrl);
  const blob = await response.blob();
  
  const tx = db.transaction('audio_store', 'readwrite');
  await tx.objectStore('audio_store').put(blob, trackId);
  console.log('Track guardado exitosamente en almacenamiento local IndexedDB');
}

// Reproducción offline desde IndexedDB
async function playOffline(trackId) {
  const db = await openDB();
  const tx = db.transaction('audio_store', 'readonly');
  const blob = await tx.objectStore('audio_store').get(trackId);
  
  const localUrl = URL.createObjectURL(blob);
  const audio = new Audio(localUrl);
  audio.play();
}

function openDB() {
  return new Promise((resolve) => {
    const req = indexedDB.open('MusicEngineCache', 1);
    req.onupgradeneeded = () => req.result.createObjectStore('audio_store');
    req.onsuccess = () => resolve(req.result);
  });
}

2. Roku SceneGraph (BrightScript Caching en Local Storage)


' BrightScript: Descarga de MP3 a la memoria local cachefs:/ del dispositivo Roku
sub DownloadTrackToRokuCache(audioUrl as String, localFilename as String)
    request = CreateObject("roUrlTransfer")
    request.SetUrl(audioUrl)
    
    localPath = "cachefs:/" + localFilename + ".mp3"
    status = request.GetToFile(localPath)
    
    if status = 200
        print "Archivo MP3 guardado exitosamente en cache local de Roku: "; localPath
        PlayRokuLocalAudio(localPath)
    end if
end sub

sub PlayRokuLocalAudio(localPath as String)
    audioNode = m.top.findNode("myAudioPlayer")
    content = CreateObject("roSGNode", "ContentNode")
    content.url = localPath
    audioNode.content = content
    audioNode.control = "play"
end sub
